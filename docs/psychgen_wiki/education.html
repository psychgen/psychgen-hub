Process education data

To facilitate further manipulation, we need to pivot to longer format. We also create a new variable with only 1-digit NUS codes (nus_simplified_code) and general educational groups following the classification from SSB (edu_group)

# Pivot longer
ssb_edu <- ssb_edu |> 
  pivot_longer(cols = starts_with(c("BU_"),ignore.case = TRUE), 
               names_to = c("var", "year"), 
               names_sep = "_", 
               values_to = "edu_code")


# Get only first digit of NUS codes
ssb_edu$nus_simplified_code <- substr(ssb_edu$edu_code, 1, 1)


# Group in educational groups 
ssb_edu <- ssb_edu |> 
  mutate(edu_group = case_when(nus_simplified_code == 0 | nus_simplified_code == 1 | nus_simplified_code == 2  ~ "Low education", 
                               nus_simplified_code == 3 | nus_simplified_code == 4 | nus_simplified_code == 5  ~ "Medium education", 
                               nus_simplified_code == 6 | nus_simplified_code == 7 | nus_simplified_code == 8  ~ "High education", 
                               nus_simplified_code == 9 ~ "Unspecified",
                               TRUE ~ NA))

We then can use the 1-digit NUS codes to get the corresponding ISCED 2011 codes (https://www.ssb.no/klass/klassifikasjoner/36/versjon/3425/korrespondanser/2862). In general:

  • ISCED 0: Early childhood education (‘less than primary’ for educational attainment)

  • ISCED 1: Primary education

  • ISCED 2: Lower secondary education

  • ISCED 3: Upper secondary education

  • ISCED 4: Post-secondary non-tertiary education

  • ISCED 5: Short-cycle tertiary education

  • ISCED 6: Bachelor’s or equivalent level

  • ISCED 7: Master’s or equivalent level

  • ISCED 8: Doctoral or equivalent level

# ISCED equivalent codes and descriptions 
nus_corresp <- readRDS(here::here("data", "correp_nus_isced.rds")) # got codes from klassR, can also download from SSB's webpage 

ssb_edu <- ssb_edu |> 
  left_join(nus_corresp, by = c("edu_code" = "nus_codes"))

ssb_edu$isced_simplified_code <- substr(ssb_edu$isced_codes, 1, 1)



ssb_edu <- ssb_edu |> 
  mutate(isced_simplified_code_description = case_when(
    isced_simplified_code == 0  ~ "Early childhood education", 
    isced_simplified_code == 1  ~ "Primary education", 
    isced_simplified_code == 2  ~ "Lower secondary education", 
    isced_simplified_code == 3  ~ "Upper secondary education", 
    isced_simplified_code == 4  ~ "Post-secondary non-tertiary education", 
    isced_simplified_code == 5  ~ "Short-cycle tertiary education", 
    isced_simplified_code == 6  ~ "Bachelor’s or equivalent level", 
    isced_simplified_code == 7  ~ "Master’s or equivalent level", 
    isced_simplified_code == 8  ~ "Doctoral or equivalent level", 
    TRUE ~ NA))

Finally, we can use the NUS codes to approximate the minimum years of completed education (https://www.ssb.no/utdanning/_attachment/283616?_ts=1583e453200, section 3.3.1):

years <- read.csv(file = here::here("data", "years_education.csv"), sep=";", dec = ",")
years$nus <- as.character(years$nus)
years$years_edu <- as.numeric(years$years_edu)


ssb_edu <- ssb_edu |> left_join(years, by = c("nus_simplified_code" = "nus"))

Merge with MBRN

We use the MBRN to get the year of birth for each child:

MBRN <- read_spss(str_subset(mbrn_path, regex('PDB2306_MBRN_541_v12', ignore_case = T))) |> 
  mutate(ind_id = ifelse(PREG_ID_2306, paste0(str_trim(PREG_ID_2306), "_", str_trim(BARN_NR)), NA_real_), 
         type = "barn")

MBRN$PREG_ID_2306 <- as.character(MBRN$PREG_ID_2306)


MBRN_filtered <- SV_INFO |> 
  left_join(MBRN |> 
              select(PREG_ID_2306, BARN_NR, ind_id, FAAR)) |> 
  select(PREG_ID_2306, BARN_NR, ind_id, M_ID_2306, F_ID_2306, FAAR)
Joining with `by = join_by(PREG_ID_2306)`
# Education at year of birth  ---------------------------------------------

ssb_edu_long_linked$year <- as.double(ssb_edu_long_linked$year)

ssb_edu_parents <- ssb_edu_long_linked |> 
  filter(type != "barn")

ssb_edu_mbrn_parents <- MBRN_filtered  |> 
  filter(!is.na(FAAR)) |> 
  left_join(
    ssb_edu_parents |> 
      filter(type == "mor") |>  
      rename(ID_mor_join = ID_2306),
    by = c("M_ID_2306" = "ID_mor_join", "FAAR" = "year")
  ) |>
  left_join(
    ssb_edu_parents |>
      filter(type == "far") |> 
      rename(ID_far_join = ID_2306),
    by = c("F_ID_2306" = "ID_far_join", "FAAR" = "year"), 
    suffix = c("_m", "_f")
  ) 


ssb_edu_mbrn_parents <- ssb_edu_mbrn_parents |> 
  select(PREG_ID_2306, BARN_NR, ind_id, M_ID_2306, F_ID_2306, FAAR, 
         nus_simplified_code_m, nus_simplified_code_f, edu_group_f,
         edu_group_m, isced_simplified_code_f, isced_simplified_code_m, isced_simplified_code_description_m,
         isced_simplified_code_description_f, years_edu_m, years_edu_f)
